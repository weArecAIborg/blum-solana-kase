3const assert = require('node:assert/strict');

9const root = path.resolve(__dirname, '../../docs/demo');
10const types = { '.html': 'text/html', '.json': 'application/json', '.js': 'application/javascript' };
11const server = http.createServer((req, res) => {
12  const file = path.resolve(root, '.' + decodeURIComponent(new URL(req.url, 'http://localhost').pathname));
13  if (!file.startsWith(root + path.sep)) { res.writeHead(403).end(); return; }
14  fs.readFile(file, (err, body) => {
15    if (err) { res.writeHead(404).end(); return; }
16    res.setHeader('Content-Type', types[path.extname(file)] || 'text/plain');
17    res.end(body);
18  });
19});
