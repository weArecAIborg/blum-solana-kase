1"""Build the stage-by-stage comparison (docs/demo/stages.html): for each stage of a bond's life, the textbook bond
2(left) beside the same stage tokenized on Solana (right), each with its own picture, and the outcome they share.

4Left side: what corporate-finance practice says, with its source (docs/tradfi-vs-onchain.md, docs/bond-mechanics.md).

9Auditor supplement required beside the output: data/stages-probe-ledger.json (cached finalized probe responses).

33def tx(href, label="transaction", evidence=None, cls="", chip=True):
34    classes = " ".join(c for c in ("bs-proof" if chip else "", "audit-link" if evidence else "", cls) if c)

53def record_tables(d, evidence):
54    """Stage 3 as the kit lays it out: the 09:00 list (textbook) and the ledger entries around it (on Solana)."""
55    after_s, after_b = d["sellerAtRecord"] - d["sold"], d["buyerAtRecord"] + d["sold"]
56    book = (f'<table class="bs-register"><thead><tr><th>Holder</th><th class="r">Bonds at 09:00</th></tr></thead><tbody>'
57            f'<tr><td>{esc(d["seller"])}</td><td class="r">{d["sellerAtRecord"]}</td></tr>'
58            f'<tr><td>{esc(d["buyer"])}</td><td class="r">{d["buyerAtRecord"]}</td></tr></tbody></table>')
59    chain = (f'<table class="bs-ledger"><thead><tr><th>Time</th><th>Entry</th><th class="r">{esc(d["seller"])}</th>'
60             f'<th class="r">{esc(d["buyer"])}</th></tr></thead><tbody>'
61             f'<tr><td>09:00:00</td><td>record moment 0</td><td class="r">{d["sellerAtRecord"]}</td><td class="r">{d["buyerAtRecord"]}</td></tr>'
62             f'<tr class="hl"><td>+{d["saleLag"]} s</td><td>hook: positions saved for 09:00</td><td class="r">{d["sellerAtRecord"]}</td>'
63             f'<td class="r">{d["buyerAtRecord"]}</td></tr>'
64             f'<tr><td>+{d["saleLag"]} s</td><td>transfer {d["sold"]} → {esc(d["buyer"])}</td><td class="r">{after_s}</td>'
65             f'<td class="r">{after_b}</td></tr></tbody></table>')
66    links = f'<p class="bs-links">{proof(evidence, d["saleKey"], "the sale")} {proof(evidence, d["payKey"], "the payment")}</p>'
67    return book, chain + links

70def receipts(d):
71    """Stage 4 as the kit lays it out: the same sum on a coupon statement and in the cash token's smallest unit."""
72    units = 10 ** d["decimals"]
73    row = lambda k, v, cls="": f'<div class="bs-rl{cls}"><span>{esc(k)}</span><span>{esc(v)}</span></div>'  # noqa: E731
74    book = ('<div class="bs-receipt book">' + row("Holding", f'{d["bonds"]} bonds') + row("Face value", f'{d["face"]:,.2f}')
75            + row("Interest", f'{d["rateBps"] / 100:g}% a year') + row("Payments a year", d["perYear"])
76            + row("Coupon", f'{d["bonds"]} × {d["face"]:,.0f} × {d["rateBps"] / 100:g}% / {d["perYear"]}')
77            + row("Rounded down to", "0.01") + row("Owed", f'{d["paid"]:,.2f}', " total") + "</div>")
78    chain = ('<div class="bs-receipt chain">' + row("position", d["bonds"]) + row("face_units", f'{d["faceUnits"]:,}')
79             + row("rate_bps", d["rateBps"]) + row("periods", d["perYear"])
80             + row("amount", f'{d["bonds"]} x {d["faceUnits"]:,} x {d["rateBps"]} / 10,000 / {d["perYear"]}')
81             + row("units", f'{round(d["paid"] * units):,}') + row("paid", f'{d["paid"]:.{d["decimals"]}f}', " total")
82             + row("QuantLib, FinancePy", f'{d["paid"]:,.2f}', " check") + "</div>")
83    return book, chain

86def screens_html(stages, evidence):
87    def refused(rows):
88        return f"""<ul class="bs-ref">{"".join(f'''<li><span class="bs-mark refused">Refused</span>
89            <span>{esc(r["actor"][0].upper() + r["actor"][1:])} {esc(r["tries"])} · {tx(r["explorer"], 'transaction', 'attack:' + r['id'])}</span></li>''' for r in rows)}</ul>""" \
90            if rows else ""
91
92    def bodies(i, s):
93        if s["id"] == "record":
94            return record_tables(s["data"], evidence)
95        if s["id"] == "calculate":
96            return receipts(s["data"])
97        return (f'<div class="viz" data-stage="{i}" data-side="left"></div>', f'<div class="viz" data-stage="{i}" data-side="right"></div>')
98
99    def sim(s):
100        if s["id"] != "record":
101            return ""
102        d = s["data"]
103        return (f'<p class="bs-sim"><b>Simulated</b><span>Devnet time is compressed: 1 bond day = {d["scale"]} chain seconds, '
104                f'so “{d["saleLag"]} s after 09:00” is {d["saleLag"]} chain seconds.</span></p>')
105
106    out = []
107    for i, s in enumerate(stages):
108        left, right = bodies(i, s)
109        out.append(f"""
110  <section class="bs-screen" id="{s["id"]}">
111    <div class="bs-sh"><span class="no">{i + 1} / {len(stages)}</span><h2>{NAMES[s["id"]]}: {esc(s["title"][0].lower() + s["title"][1:])}</h2></div>
112    <div class="bs-lr">
113      <div class="bs-side book"><span class="label">Textbook</span>
114        {left}
115        <p class="cap">{esc(s["left"]["caption"])}</p><p class="src">{esc(s["left"]["source"])} ·
116          {" · ".join(tx(url, name, 'rule:' + s['id'], chip=False) for name, url in RULES[s['id']])}</p></div>
117      <div class="bs-side chain"><span class="label">On Solana, measured</span>
118        {right}
119        <p class="cap">{esc(s["right"]["caption"])}</p>
120        <div class="bs-value">{proof(evidence, 'stage:' + s['id'], s['right']['value'], 'metric-link', chip=False)}</div>
121        {refused(s["right"]["refused"])}
122        <p class="bs-links">{proof(evidence, 'stage:' + s['id'], 'inspect evidence')}
123          {tx('data/devnet-crowd-1000.json', 'measured data', 'stage:' + s['id'])}
124          {proof(evidence, 'keeper', 'outside keeper') if s['id'] == 'pay' else ''}
125          {tx('data/devnet-hostile-attempts.json', f'all {s["right"]["probes"]} probes', 'stage:audit') if s['id'] == 'audit' else ''}</p></div>
126    </div>
127    <p class="bs-now ok"><b>Same outcome</b><span>{esc(s["verdict"])}</span></p>
128    {sim(s)}
129  </section>""")
130    return "".join(out)

142def fill(template, parts):
143    for key, html in parts.items():
144        marker = f"<!--__{key}__-->" if key != "PAGE" else "/*__PAGE__*/null"
145        assert template.count(marker) == 1, f"template must hold {marker} once"
146        template = template.replace(marker, html)

150def main(crowd_path, keeper_path, probe_path, hostile_path, out_path):
151    D = json.loads(Path(crowd_path).read_text())
152    K = json.loads(Path(keeper_path).read_text())
153    probe = json.loads(Path(probe_path).read_text())
154    H = json.loads(Path(hostile_path).read_text())
155    scn = json.loads((ROOT / D["scenarioFile"]).read_text())
156    T = scn["terms"]
157    unit = 10 ** T["cashDecimals"]
158    face = int(T["faceUnits"]) / unit
159    per_bond = D["comparison"]["perBond"]
160    c1, c5 = card_market_open(D), card_audit(D, probe)
161    att = {a["id"]: a for a in H["attempts"]}
162    failing = [c["id"] for c in D["audit"]["checks"] if not c["ok"]] + [a["id"] for a in H["attempts"] if not a["byGuard"]]
163    if failing or c1["exact"] != c1["cells"]:
164        raise SystemExit(f"not building: failing {failing}, exact {c1['exact']} of {c1['cells']}")
165
166    records = [r["chainTime"] for r in D["records"]]
167    events = D["events"]
168    tender = next(o for o in scn["offers"] if o["kind"] == "tender")
169    buyback = next(o for o in scn["offers"] if o["kind"] == "buyback")
170    tender_price = int(tender["priceUnits"]) / unit
171
172    # Bonds over the bond's life (supply) and what retired them.
173    supply, s_t, s_v, retired = 0, [], [], {"buyback": 0, "tender": 0, "redemption": 0}
174    for e in events:
175        change = sum(m["bonds"] for m in e["moves"].values()) if e["kind"] != "trade" else 0
176        if not change:
177            continue
178        supply += change
179        s_t.append(e["t"])
180        s_v.append(supply)
181        if e["kind"] == "buyback":
182            for m in e["moves"].values():
183                if m["bonds"]:
184                    retired["tender" if round(m["cash"] / -m["bonds"], 6) == tender_price else "buyback"] -= m["bonds"]
185        elif e["kind"] == "redemption":
186            retired["redemption"] -= change
187    placed = sum(m["bonds"] for e in events if e["kind"] == "placement" for m in e["moves"].values())
188
189    # Trades: seconds after the record moment before them (bond days at the run's scale).
190    lags = []
191    for e in events:
192        if e["kind"] == "trade":
193            passed = [r for r in records if r <= e["t"]]
194            if passed:
195                lags.append(round(e["t"] - passed[-1], 1))
196
197    # Coupon 0: every payment transaction, seconds after the first one; holders paid in each.
198    c0 = [e for e in events if e["kind"] == "coupon" and e["coupon"] == 0]
199    t_first = min(e["t"] for e in c0)
200    pay_batches = sorted((e["t"] - t_first, sum(1 for m in e["moves"].values() if m["cash"] > 0)) for e in c0)
201    supply0 = sum(D["grid"][0]["atRecord"].values())
202
203    # The listing's example holder through redemption.
204    bh1_red = next(e for e in events if e["kind"] == "redemption" and "BH#1" in e["moves"])
205    # Tender: who tendered (the scenario's instructions), what was accepted (measured), coupon in the window.
206    tendered_by = {}
207    for a in tender["actions"]:
208        tendered_by[a["holder"]] = tendered_by.get(a["holder"], 0) + a.get("tender", 0) - a.get("withdraw", 0)
209    tendered_by = {h: q for h, q in tendered_by.items() if q > 0}
210    accepted_by = {}
211    for e in events:
212        if e["kind"] == "buyback":
213            for h, m in e["moves"].items():
214                if m["bonds"] and round(m["cash"] / -m["bonds"], 6) == tender_price:
215                    accepted_by[h] = accepted_by.get(h, 0) - m["bonds"]
216    window = tender["close"]["recordMoment"]
217    kept_coupon = sum(1 for h in tendered_by if D["grid"][window]["paid"].get(h, 0) > 0)
218    total_tendered = sum(tendered_by.values())
219    holders_tender = sorted(tendered_by, key=lambda h: -tendered_by[h])
220
221    keeper_exact = sum(1 for g in K["grid"] for n in g["expected"] if abs(g["paid"][n] - g["expected"][n]) < 1e-9)
222    keeper_cells = sum(len(g["expected"]) for g in K["grid"])
223    serviced = [s for s in H["serviced"] if s["cashTo"]]
224    evidence, payment_events, sources = build_evidence(D, K, probe, H, scn, Path(out_path).parent)
225

340                   "refused": [], "probes": len(H["attempts"])},
341         "verdict": f"Same books, open to anyone: {c5['holdersExact']:,} of {c5['holders']:,} holders' cash equals the "
342                    f"terms to the unit.",
343         "draw": "checks", "data": {"passed": c5["passed"], "checks": c5["checks"], "probeFailed": len(probe["after"]["failed"]),
344                                    "probeChecks": probe["after"]["checks"]}},
345    ]
346    for stage in stages:
347        if stage["id"] == "trade":
348            stage["data"]["tradeKeys"] = [f'tx:{e["signature"]}' for e in events if e["kind"] == "trade" and any(r <= e["t"] for r in records)]
349        elif stage["id"] == "pay":
350            stage["data"]["cells"] = [{"t": e["t"] - t_first, "key": f"paid:{name}"} for e in sorted(payment_events, key=lambda x: x["t"])
351                                      for name, m in e["moves"].items() if m["cash"] > 0]
352        elif stage["id"] == "offer":
353            stage["data"]["holders"] = holders_tender
354        elif stage["id"] == "record":
355            stage["data"]["saleKey"] = next(key for key, value in evidence.items() if key.startswith("tx:") and value["refs"][0]["href"] == c1["saleLink"])
356            stage["data"]["payKey"] = next(key for key, value in evidence.items() if key.startswith("tx:") and value["refs"][0]["href"] == c1["payLink"])
357    # Embed only evidence used by this page; the full registry remains available as a linked artifact.
358    keys = {"program", "keeper"} | {f"{kind}:{s['id']}" for s in stages for kind in ("stage", "rule")}
359    keys |= {f"attack:{a['id']}" for a in H["attempts"]}
360    keys |= {key for s in stages for key in s.get("data", {}).get("tradeKeys", [])}
361    keys |= {c["key"] for s in stages for c in s.get("data", {}).get("cells", [])}
362    keys |= {f"offer:{name}" for name in holders_tender}
363    keys |= {f"check:{i}" for i in range(len(D["audit"]["checks"]))}
364    keys |= {s["data"][key] for s in stages for key in ("saleKey", "payKey") if key in s.get("data", {})}
365    keys |= {evidence[key]["related"] for key in list(keys) if evidence[key].get("related")}
366    previews = {key: evidence[key] for key in sorted(keys)}
367    # Repeated source/data URLs are stored once, rather than once per holder/assertion.
368    references, ref_ids = [], {}
369    for key, entry in previews.items():
370        entry = dict(entry)
371        ids = []
372        for r in entry["refs"]:
373            identity = (r["label"], r["href"])
374            if identity not in ref_ids:
375                ref_ids[identity] = len(references)
376                references.append(r)
377            ids.append(ref_ids[identity])
378        entry["refs"] = ids
379        previews[key] = entry
380    page = {"stages": stages, "programLink": D["programLink"], "generated": D["generated"],
381            "terms": {"face": face, "ratePct": T["rateBps"] / 100, "perYear": T["periodsPerYear"], "issue": T["issue"],
382                      "maturity": T["maturity"], "payDates": [g["payDate"] for g in D["grid"]]},
383            "serviced": len(serviced), "evidence": previews, "references": references}
384    template = (ROOT / "tools/demo/stages_template.html").read_text()
385    safe_json = json.dumps(page, ensure_ascii=False).replace("<", "\\u003c").replace(">", "\\u003e").replace("&", "\\u0026")
386    html = fill(template, {"INDEX": index_html(stages), "SCREENS": screens_html(stages, evidence),
387                           "FOOT": foot_html(page), "PAGE": safe_json})
388    Path(out_path).write_text("\n".join(line.rstrip() for line in html.splitlines()) + "\n")
389    print(f"wrote {out_path}: {len(stages)} stages")
