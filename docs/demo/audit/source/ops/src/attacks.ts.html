1// Hostile actor for a bond run (scenario field "attacker"). At each step of the bond's life it tries the actions that
2// would take bonds or cash from someone, or let someone act without authority. Every attempt is sent without
3// simulation (skipPreflight), so a refusal is a failed transaction on the chain that anyone can open in an explorer.
4// An attempt passes only if it is refused by the guard named in `guard`; the run fails otherwise
5// (scenario.ts exits with an error after writing the report).
6// Also shows the bank's frozen cash account case: one holder cannot be paid, the others are paid, nobody waits.
7

92// Why a transaction failed, from its logs: the Anchor error code, a token program's error text, or the runtime's.
93const reasonOf = (meta: any) => {
94  const logs: string[] = meta.logMessages ?? [];
95  for (const l of logs) {
96    const m = /Error Code: (\w+)\. Error Number: \d+\. Error Message: (.*)$/.exec(l);
97    if (m) return `${m[1]}: ${m[2]}`;
98  }
99  for (const l of logs) {
100    const m = /^Program log: Error: (.*)$/.exec(l);
101    if (m) return m[1];
102  }
103  const failed = logs.map((l) => /^Program \w+ failed: (.*)$/.exec(l)).filter(Boolean).pop();
104  return failed ? failed[1] : JSON.stringify(meta.err);
105};

107// A signer the instruction names but that nobody can sign for (a program address): keep it in the instruction as a
108// non-signer, so the transaction reaches the chain and the program checking the signature refuses it there.
109const unsigned = (ix: TransactionInstruction, key: PublicKey) => {
110  for (const k of ix.keys) if (k.pubkey.equals(key)) k.isSigner = false;
111  return ix;
112};

114export function hostile(c: Ctx) {
115  const attempts: Attempt[] = [];
116  const A = c.attacker.publicKey;
117  const victim = c.spec.victim;
118  const accomplice = c.spec.accomplice;
119  const attackerBonds = getAssociatedTokenAddressSync(c.mint, A, false, TOKEN_2022_PROGRAM_ID);
120  const attackerCash = getAssociatedTokenAddressSync(c.cashMint, A, false, TOKEN_PROGRAM_ID);
121  const victimSecondAccount = Keypair.generate();
122  const asAttacker = new anchor.Program(c.program.idl, new anchor.AnchorProvider(c.connection, new anchor.Wallet(c.attacker), { commitment: "confirmed" }));

124  // Send without simulation and poll its status until it lands (failed or not); resend with a fresh blockhash if it
125  // expired first. (web3.js confirmTransaction throws a landed transaction's error on devnet.)
126  const land = async (ixs: TransactionInstruction[], signers: Keypair[]) => {
127    for (let tries = 1; tries <= 4; tries++) {
128      const { blockhash, lastValidBlockHeight } = await c.connection.getLatestBlockhash("confirmed");
129      const tx = new Transaction({ feePayer: signers[0].publicKey, blockhash, lastValidBlockHeight }).add(...ixs);
130      tx.sign(...signers);
131      const signature = await c.connection.sendRawTransaction(tx.serialize(), { skipPreflight: true, maxRetries: 5 });
132      for (;;) {
133        const st = (await c.connection.getSignatureStatuses([signature])).value[0];
134        if (st?.confirmationStatus === "confirmed" || st?.confirmationStatus === "finalized") return signature;
135        if ((await c.connection.getBlockHeight("confirmed")) > lastValidBlockHeight) break;
136        await new Promise((r) => setTimeout(r, 1_000));
137      }
138    }
139    throw new Error("attempt did not land after 4 sends");
140  };
141  const attempt = async (
142    a: { id: string; when: string; actor: string; tries: string; wouldCost: string; guard: string },
143    ixs: TransactionInstruction[],
144    signers: Keypair[],
145  ) => {
146    const signature = await land(ixs, signers);
147    let tx = null;
148    for (let i = 0; i < 20 && !tx; i++) {
149      tx = await c.connection.getTransaction(signature, { commitment: "confirmed", maxSupportedTransactionVersion: 0 });
150      if (!tx) await new Promise((r) => setTimeout(r, 1_000));
151    }
152    if (!tx) throw new Error(`${a.id}: transaction ${signature} not found`);
153    const outcome = tx.meta!.err ? "refused" : "accepted";
154    const reason = tx.meta!.err ? reasonOf(tx.meta) : "";
155    const ok = outcome === "refused" && reason.includes(a.guard);
156    const row: Attempt = { ...a, outcome, reason, ok, signature, explorer: c.explorer(signature), chainTime: new Date((await c.chainNow()) * 1000).toISOString() };
157    attempts.push(row);
158    console.log(`${ok ? "REFUSED " : "FAILED  "} ${a.id}: ${a.tries} -> ${outcome}${reason ? ` (${reason})` : ""}  ${row.explorer ?? signature}`);
159    return row;
160  };

171      createAssociatedTokenAccountIdempotentInstruction(A, attackerCash, A, c.cashMint, TOKEN_PROGRAM_ID),
172      SystemProgram.createAccount({

174        lamports: await c.connection.getMinimumBalanceForRentExemption(size), programId: TOKEN_2022_PROGRAM_ID,

344  // Payment is open; before anyone is paid. Returns the holders to pay after afterPayments.

357      await attempt({ id: "other-vault", when, actor: "attacker", tries: `pays ${victim}'s coupon 1 from coupon 0's vault`,
358        wouldCost: "one coupon's cash spent on another", guard: "ConstraintHasOne" },
359        [await payCoupon(asAttacker, A, e, victim, { vault: c.pda(Buffer.from("vault"), prev.toBuffer()) })], [c.attacker]);
360    }
361    if (e !== c.spec.frozenCash.coupon) return [];
362    const who = c.spec.frozenCash.holder;
363    await c.send(`bank freezes ${who}'s cash account`, [createFreezeAccountInstruction(c.cashAta(who), c.cashMint, c.bank.publicKey, [], TOKEN_PROGRAM_ID)], [c.registrar, c.bank]);
364    return [who];
365  };
366
367  // Everyone else is paid.
368  const afterPayments = async (e: number) => {
369    const when = `coupon ${e}, paid`;
370    if (e === 0) {
371      await attempt({ id: "pay-twice", when, actor: "attacker", tries: `pays ${victim}'s coupon again`,
372        wouldCost: "the issuer pays a coupon twice; the last holders find the vault empty", guard: "AlreadyPaid" },
373        [await payCoupon(asAttacker, A, e, victim)], [c.attacker]);
374      const recs: any[] = await c.program.account.holderRecord.fetchMultiple(Object.keys(c.holders).map(c.holderRecord));
375      const late = Object.keys(c.holders).find((n, i) => recs[i] && Number(recs[i].position.liveFrom) > e && Number(recs[i].position.at[e]) === 0 && Number(recs[i].position.quantity) > 0);
376      if (!late) throw new Error("scenario has no holder that bought after record moment 0 with none before (needed by attack late-buyer)");
377      await attempt({ id: "late-buyer", when, actor: late, tries: "bought after the record moment and asks for the coupon",
378        wouldCost: "the seller's coupon paid again to the buyer", guard: "NothingToPay" },
379        [await payCoupon(c.program, c.registrar.publicKey, e, late)], [c.registrar]);
380    }
381    if (e !== c.spec.frozenCash.coupon) return;
382    const who = c.spec.frozenCash.holder;
383    await attempt({ id: "frozen-cash", when, actor: "anyone", tries: `pays ${who}, whose cash account the bank froze`,
384      wouldCost: "nothing: the others are already paid, and the right stays open", guard: "Account is frozen" },
385      [await payCoupon(c.program, c.registrar.publicKey, e, who)], [c.registrar]);
386    await c.send(`bank thaws ${who}'s cash account`, [createThawAccountInstruction(c.cashAta(who), c.cashMint, c.bank.publicKey, [], TOKEN_PROGRAM_ID)], [c.registrar, c.bank]);
387  };
388
389  // After a phase's buyback sales and tenders.
390  const afterOfferActions = async (e: number) => {
391    for (const o of c.offers) {
392      if (!o.actions.some((a) => a.afterRecordMoment === e)) continue;
393      const when = `${o.kind} offer ${o.id} open`;
394      if (o.kind === "buyback") {
395        await attempt({ id: "sell-others-bonds", when, actor: "attacker", tries: `sells ${victim}'s bonds to the buyback`,
396          wouldCost: `${victim}'s bonds sold without its consent`, guard: "ConstraintRaw" },
397          [await asAttacker.methods.sellToOffer(new BN(1)).accountsPartial({ retirement: c.retirementAccounts(o, victim, A), seller: A, offer: c.offerPda(o.id) }).instruction()], [c.attacker]);
398        await attempt({ id: "over-maximum", when, actor: victim, tries: "sells 1 bond to the buyback after its maximum is reached",
399          wouldCost: "the issuer buys more bonds than it offered", guard: "ExceedsOffer" },
400          [await c.program.methods.sellToOffer(new BN(1)).accountsPartial({ retirement: c.retirementAccounts(o, victim, c.holders[victim].publicKey), seller: c.holders[victim].publicKey, offer: c.offerPda(o.id) }).instruction()],
401          [c.registrar, c.holders[victim]]);
402      } else {
403        const tenderer = o.actions.find((a) => a.afterRecordMoment === e && a.tender)!.holder;
404        const rec: any = await c.program.account.holderRecord.fetch(c.holderRecord(tenderer));
405        const to = tenderer === accomplice ? victim : accomplice;
406        await attempt({ id: "sell-tendered", when, actor: tenderer, tries: `transfers all its bonds to ${to}, including those tendered to the issuer`,
407          wouldCost: "the same bonds sold twice: to the issuer and to a buyer", guard: "BondsLocked" },
408          [await hooked(tenderer, c.holders[tenderer], c.bondsAta(to), Number(rec.position.quantity))], [c.registrar, c.holders[tenderer]]);
409      }
410    }
411  };
412
413  // Trading has closed for redemption.
414  const afterFinalRecordMoment = async () => {
415    await attempt({ id: "trade-at-maturity", when: "after the final record moment", actor: accomplice, tries: `sells 1 bond to ${victim}`,
416      wouldCost: "a bond changes hands while its principal is being paid", guard: "TransfersClosedAtMaturity" },
417      [await hooked(accomplice, c.holders[accomplice], c.bondsAta(victim), 1)], [c.registrar, c.holders[accomplice]]);
418  };
419
420  const redeemIx = (sv: anchor.Program, payer: PublicKey, event: PublicKey, vault: PublicKey, name: string, over: Record<string, PublicKey> = {}) =>
421    sv.methods.redeemHolder().accountsPartial({
422      payer, bond: c.bond, mint: c.mint, event, holder: c.holderRecord(name), owner: c.holders[name].publicKey, holderBonds: c.bondsAta(name),
423      vault, cashMint: c.cashMint, holderCash: c.cashAta(name), memoProgram: null, tokenProgram: TOKEN_2022_PROGRAM_ID, cashTokenProgram: TOKEN_PROGRAM_ID, ...over,
424    }).instruction();
425
426  const beforeRedemption = async (event: PublicKey, vault: PublicKey) => {
427    const when = "redemption, payment open";
428    await attempt({ id: "redirect-principal", when, actor: "attacker", tries: `redeems ${victim}'s bonds into its own cash account`,
429      wouldCost: `${victim}'s principal`, guard: "ConstraintTokenOwner" },
430      [await redeemIx(asAttacker, A, event, vault, victim, { holderCash: attackerCash })], [c.attacker]);
431    await attempt({ id: "coupon-from-principal", when, actor: "attacker", tries: `pays ${victim} a coupon from the redemption vault`,
432      wouldCost: "principal cash paid out as coupons", guard: "WrongEventKind" },
433      [await asAttacker.methods.payCoupon().accountsPartial({
434        payer: A, bond: c.bond, event, holder: c.holderRecord(victim), owner: c.holders[victim].publicKey, vault, cashMint: c.cashMint,
435        holderCash: c.cashAta(victim), memoProgram: null, cashTokenProgram: TOKEN_PROGRAM_ID,
436      }).instruction()], [c.attacker]);
437  };
438
439  const afterRedemption = async (event: PublicKey, vault: PublicKey) => {
440    await attempt({ id: "redeem-twice", when: "redemption, paid", actor: "attacker", tries: `redeems ${victim} again`,
441      wouldCost: "principal paid twice", guard: "NothingToPay" },
442      [await redeemIx(asAttacker, A, event, vault, victim)], [c.attacker]);
443  };
444
445  return { attempts, beforeFirstRecordMoment, afterRecordMoment, whileUnfunded, beforePayments, afterPayments, afterOfferActions, afterFinalRecordMoment, beforeRedemption, afterRedemption };
446}
