1"""Measure a hostile run from the ledger, independently of the operator's report.
2
3The operator's report (ops/out/devnet-hostile.json, written by ops/src/scenario.ts with ops/src/attacks.ts) says which
4attempts were made: who, what, what it would cost, which guard should stop it, and the transaction signature. Here
5every signature is fetched again from public RPC:
6  - an attempt is refused if its transaction failed; the reason is read from the transaction's own logs;
7  - the attacker wallet's successful transactions are listed with every token balance they changed, so the page can
8    show whose accounts received cash from them.
9
10Usage: .venv/bin/python3 tools/demo/measure_attempts.py ops/out/devnet-hostile.json https://api.devnet.solana.com \
11           docs/demo/data/devnet-hostile-attempts.json
12"""

14import contextlib
15import io
16import json
17import re
18import sys
19from datetime import datetime, timezone
20from pathlib import Path

22ROOT = Path(__file__).resolve().parents[2]
23sys.path.insert(0, str(ROOT / "tools/audit"))
24import audit as A  # noqa: E402

27def reason_of(meta):
28    """Why a transaction failed: the Anchor error code, a token program's error text, or the runtime's."""
29    logs = meta.get("logMessages") or []
30    for line in logs:
31        m = re.search(r"Error Code: (\w+)\. Error Number: \d+\. Error Message: (.*)$", line)
32        if m:
33            return f"{m.group(1)}: {m.group(2)}"
34    for line in logs:
35        m = re.match(r"^Program log: Error: (.*)$", line)
36        if m:
37            return m.group(1)
38    failed = [m.group(1) for line in logs if (m := re.match(r"^Program \w+ failed: (.*)$", line))]
39    return failed[-1] if failed else json.dumps(meta["err"])

42def token_changes(tx):
43    """Token balance changes of a transaction: (owner, mint, change in base units)."""
44    meta = tx["meta"]
45    pre = {(b["accountIndex"]): b for b in meta.get("preTokenBalances") or []}
46    post = {(b["accountIndex"]): b for b in meta.get("postTokenBalances") or []}
47    out = []
48    for i in set(pre) | set(post):
49        b = post.get(i) or pre[i]
50        before = int(pre[i]["uiTokenAmount"]["amount"]) if i in pre else 0
51        after = int(post[i]["uiTokenAmount"]["amount"]) if i in post else 0
52        if after != before:
53            out.append({"owner": b["owner"], "mint": b["mint"], "change": after - before})
54    return out

57def main(report_path, rpc_url, out_path):
58    report = json.loads(Path(report_path).read_text())
59    scn = json.loads((ROOT / "scenarios" / f"{report['scenario']}.json").read_text())
60    rpc = A.Rpc(rpc_url)
61    get = lambda sig: rpc.call("getTransaction", [sig, {"encoding": "jsonParsed", "maxSupportedTransactionVersion": 0,
62                                                        "commitment": "confirmed"}])
63    attacker = A.pubkey_of(scn["attacker"]["wallet"])
64    holders = {A.pubkey_of(path): name for name, path in scn["wallets"].items()}
65
66    attempts = []
67    for a in report["attempts"]:
68        tx = get(a["signature"])
69        failed = tx["meta"]["err"] is not None
70        reason = reason_of(tx["meta"]) if failed else ""
71        attempts.append({k: a.get(k) for k in ("id", "when", "actor", "tries", "wouldCost", "guard", "signature", "explorer")}
72                        | {"refused": failed, "reason": reason, "byGuard": failed and a["guard"] in reason,
73                           "blockTime": tx["blockTime"]})
74
75    # The attacker wallet's successful transactions: every one the operator logged with the attacker as fee payer.
76    serviced = []
77    for step in report["transactions"]:
78        tx = get(step["signature"])
79        keys = tx["transaction"]["message"]["accountKeys"]
80        if keys[0]["pubkey"] != attacker or tx["meta"]["err"] is not None:
81            continue
82        changes = token_changes(tx)
83        serviced.append({
84            "step": step["step"], "signature": step["signature"], "explorer": step.get("explorer"),
85            "cashTo": sorted({holders.get(c["owner"], c["owner"]) for c in changes if c["mint"] == report["cashMint"] and c["change"] > 0}),
86            "cashToAttacker": sum(c["change"] for c in changes if c["mint"] == report["cashMint"] and c["owner"] == attacker and c["change"] > 0),
87            "bondsToAttacker": sum(c["change"] for c in changes if c["mint"] == report["bondMint"] and c["owner"] == attacker and c["change"] > 0),
88        })
89
90    # The independent audit of the same run (its own measurement; failed transactions change no balance).
91    with contextlib.redirect_stdout(io.StringIO()):
92        audit_exit = A.main(str(ROOT / "scenarios" / f"{report['scenario']}.json"), rpc_url, "history-accounts")
93    audit = {"passed": audit_exit == 0, "checks": [{"id": cid, "label": label, "ok": ok} for cid, label, ok in A.results],
94             "transactions": A.last_audit["transactions"]}
95
96    out = {
97        "scenario": report["scenario"], "name": report["name"], "bondMint": report["bondMint"], "cashMint": report["cashMint"],
98        "attacker": attacker, "victim": scn["attacker"]["victim"], "frozenCash": scn["attacker"]["frozenCash"],
99        "attempts": attempts, "serviced": serviced, "audit": audit,
100        "measured": datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC"), "rpc": rpc_url,
101    }
102    Path(out_path).parent.mkdir(parents=True, exist_ok=True)
103    Path(out_path).write_text(json.dumps(out, indent=1) + "\n")
104    refused = sum(1 for a in attempts if a["refused"])
105    by_guard = sum(1 for a in attempts if a["byGuard"])
106    print(f"wrote {out_path}: {refused} of {len(attempts)} attempts refused ({by_guard} by their named guard); "
107          f"attacker wallet: {len(serviced)} successful transactions, cash to itself "
108          f"{sum(s['cashToAttacker'] for s in serviced)}, bonds to itself {sum(s['bondsToAttacker'] for s in serviced)}; "
109          f"audit {sum(1 for c in audit['checks'] if c['ok'])} of {len(audit['checks'])}")
110
111
112if __name__ == "__main__":
113    if len(sys.argv) != 4:
114        print("usage: measure_attempts.py <ops/out/devnet-NAME.json> <rpc url> <out.json>", file=sys.stderr)
115        sys.exit(2)
116    main(*sys.argv[1:])
