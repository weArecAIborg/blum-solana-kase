1"""Check an operator report (ops/src/scenario.ts) against QuantLib and FinancePy, independently of the program.
2
3For every coupon, every offer (buyback, tender) and the redemption it recomputes:
4  - payment dates (QuantLib schedule + the bond's calendar as the report states it);
5  - each holder's position at each record moment, from the scenario's placements, trades and offer actions alone
6    (tools/audit/audit.py `timeline`, the same order the operator follows);
7  - per-bond coupon (FinancePy CouponType.FIXED = Kazakh decree formula; QuantLib 30/360 shown alongside);
8  - each entitlement, payment and principal, with SCoRE Standard 4 rounding (round down on the total holding);
9  - final balances.
10Run: .venv/bin/python3 tools/reference/check_report.py ops/out/<scenario>.json
11Exit code 1 on any mismatch.
12"""

23from financepy.utils.frequency import FrequencyTypes

32def check(label, ok, detail=""):
33    print(f"  [{'PASS' if ok else 'FAIL'}] {label}{(' - ' + detail) if detail else ''}")
34    if not ok:
35        failures.append(label)

38def qd(s):
39    y, m, d = map(int, s.split("-"))
40    return ql.Date(d, m, y)
41
42
43def iso(d):
44    return f"{d.year():04d}-{d.month():02d}-{d.dayOfMonth():02d}"
45

47def bond_calendar(calendar_file, weekend):
48    """The bond's calendar as the report states it (holiday file, weekend days)."""
49    cal = ql.BespokeCalendar("bond")
50    for day in weekend:
51        cal.addWeekend(getattr(ql, day))
52    for line in (ROOT / calendar_file).read_text().splitlines():
53        if line and not line.startswith("#"):
54            cal.addHoliday(qd(line))
55    return cal

58def main(report_path):
59    r = json.loads(Path(report_path).read_text())
60    t = r["terms"]
61    face_units = int(t["faceUnits"])
62    unit = 10 ** t["cashDecimals"]
63    face = Decimal(face_units) / unit
64    rate = Decimal(t["rateBps"]) / 10_000
65    per_year = 12 // t["monthsPerPeriod"]
66    print(f"{r['name']}\nbond {r['bond']}  program {r['programId']}")
67
68    print("\nSchedule")
69    cal = bond_calendar(t["calendar"], t["weekend"])
70    sched = ql.Schedule(qd(t["issue"]), qd(t["maturity"]), ql.Period(t["monthsPerPeriod"], ql.Months), cal,
71                        ql.Unadjusted, ql.Unadjusted, ql.DateGeneration.Backward, False)
72    ql_bond = ql.FixedRateBond(0, float(face), sched, [float(rate)], ql.Thirty360(ql.Thirty360.BondBasis),
73                               ql.Following, 100.0, qd(t["issue"]), cal)
74    ql_coupons = [c for c in (ql.as_fixed_rate_coupon(cf) for cf in ql_bond.cashflows()) if c is not None]
75    ql_dates = [iso(c.date()) for c in ql_coupons]
76    check("payment dates = QuantLib (bond calendar, Following)", ql_dates == t["paymentDates"],
77          ", ".join(t["paymentDates"]))
78    freq = {1: FrequencyTypes.ANNUAL, 2: FrequencyTypes.SEMI_ANNUAL}[per_year]
79    y, m, d = map(int, t["issue"].split("-"))
80    y2, m2, d2 = map(int, t["maturity"].split("-"))
81    fp = FpBond(FpDate(d, m, y), FpDate(d2, m2, y2), float(rate), freq, DayCountTypes.THIRTY_360_BOND,
82                cpn_type=CouponType.FIXED)
83    fp_flows = fp.flow_amounts[1:]
84
85    # Positions from placements, trades and offer actions only (no program data).
86    record_times = [record_moment_unix(d, t["recordTimeLocal"], t["utcOffsetHours"]) for d in t["recordDates"]]
87    tl = timeline(r["placements"], r["trades"], r["offerSpecs"], record_times)
88    holders = [x["holder"] for x in r["events"][0]["positions"]]
89    expected_cash = {h: Decimal(0) for h in holders}
90    expected_vault_left = {}
91
92    print("\nCoupons")
93    for ev in r["events"]:
94        e = ev["index"]
95        fp_per_bond = Decimal(str(fp_flows[e])) * face_units  # FinancePy FIXED, base units
96        ql_per_bond = Decimal(str(ql_coupons[e].amount())) * unit
97        positions = {h: tl["at_record"][e].get(h, 0) for h in holders}
98        supply = sum(positions.values())
99        print(f" coupon {e}: record {ev['recordDate']} {ev['recordTime']}, paid from {ev['paymentDate']}")
100        exp_budget = (Decimal(supply) * fp_per_bond).to_integral_value(rounding=ROUND_FLOOR)
101        check(f"coupon {e}: supply at record moment", ev["supplyAtRecord"] == supply, f"program {ev['supplyAtRecord']}, expected {supply}")
102        check(f"coupon {e}: budget = whole supply x FinancePy FIXED coupon, rounded down", Decimal(ev["budget"]) == exp_budget,
103              f"program {Decimal(ev['budget']) / unit}, expected {exp_budget / unit} (QuantLib 30/360 per bond {ql_per_bond / unit:.6f})")
104        paid = {p["holder"]: p for p in ev["payments"]}
105        program_pos = {x["holder"]: x for x in ev["positions"]}
106        paid_total = Decimal(0)
107        for h, exp_pos in positions.items():
108            got_pos = program_pos[h]["position"]
109            exp_amt = (Decimal(exp_pos) * fp_per_bond).to_integral_value(rounding=ROUND_FLOOR)
110            check(f"coupon {e} {h}: position at record moment", got_pos == exp_pos,
111                  f"program {got_pos}, expected {exp_pos} (balance now {program_pos[h]['balanceNow']})")
112            got = Decimal(paid[h]["amount"]) if h in paid else Decimal(0)
113            check(f"coupon {e} {h}: paid on chain", got == exp_amt,
114                  f"{got / unit} vs {exp_amt / unit}" + (f" (tx {paid[h]['signature'][:16]}...)" if h in paid else ""))
115            expected_cash[h] += exp_amt
116            paid_total += exp_amt
117        expected_vault_left[ev["vault"]] = exp_budget - paid_total
118
119    print("\nOffers (buyback, tender)")
120    measured = {h: [] for h in holders}
121    for o in r["offers"]:
122        paid, bought = Decimal(0), 0
123        for x in [a for a in o["actions"] if a["action"] == "sell"] + o["settlements"]:
124            if x.get("burned"):
125                measured[x["holder"]].append((x["burned"], int(x["paid"])))
126                paid += Decimal(x["paid"])
127                bought += x["burned"]
128        expected_vault_left[o["vault"]] = Decimal(o["budget"]) - paid
129        print(f" offer {o['id']} ({o['kind']}): price {Decimal(o['priceUnits']) / unit}, max {o['maxQuantity']}, "
130              f"window {o['openTime']} - {o['closeTime']}, bought {bought}")
131    for h in holders:
132        exp = sorted(tl["retire"][h])
133        got = sorted(measured[h])
134        check(f"offers {h}: (bonds bought, cash paid)", got == exp, f"program {got} vs expected {exp}")
135        expected_cash[h] += sum(Decimal(c) for _, c in exp)
136    positions = {h: tl["final"].get(h, 0) for h in holders}
137
138    print("\nRedemption")
139    red = {x["holder"]: x for x in r["redemption"]["redemptions"]}
140    price = Decimal(t["redemptionPriceBps"]) / 10_000
141    supply = sum(positions.values())
142    exp_budget = (Decimal(supply) * face_units * price).to_integral_value(rounding=ROUND_FLOOR)
143    check("redemption: budget = whole supply x face x price, rounded down", Decimal(r["redemption"]["budget"]) == exp_budget,
144          f"{Decimal(r['redemption']['budget']) / unit} vs {exp_budget / unit}")
145    paid_total = Decimal(0)
146    for h, pos in positions.items():
147        if pos == 0:
148            continue
149        exp_principal = (Decimal(pos) * face_units * price).to_integral_value(rounding=ROUND_FLOOR)
150        got = red.get(h)
151        check(f"redemption {h}: burned = position", got is not None and got["burned"] == pos,
152              f"{got['burned'] if got else 'none'} vs {pos}")
153        check(f"redemption {h}: principal", got is not None and Decimal(got["principal"]) == exp_principal,
154              f"{(Decimal(got['principal']) / unit) if got else 'none'} vs {exp_principal / unit}")
155        expected_cash[h] += exp_principal
156        paid_total += exp_principal
157    expected_vault_left[r["redemption"]["vault"]] = exp_budget - paid_total
158
159    print("\nFinal balances")
160    check("bond supply = 0", r["final"]["supply"] == "0")
161    for v, left in expected_vault_left.items():
162        got = Decimal(r["final"]["vaults"][v])
163        check(f"vault {v[:8]}...: left = budget - payments (rounding only)", got == left, f"{got / unit} vs {left / unit}")
164    # What this bond delivered = final cash - cash before the run (holders may hold the cash token from elsewhere).
165    initial = r.get("initialCash")
166    check("report carries initialCash", initial is not None, "" if initial is not None else "report predates initialCash; rerun the scenario")
167    for h, exp in expected_cash.items():
168        if initial is None:
169            continue
170        got = Decimal(r["final"]["cash"][h]) - Decimal(initial[h])
171        check(f"cash {h}", got == exp, f"{got / unit} vs {exp / unit} (final {Decimal(r['final']['cash'][h]) / unit}, "
172                                       f"initial {Decimal(initial[h]) / unit})")
173
174    if r["scenario"] == "b1":
175        print("\nListing example")
176        first = {x["holder"]: x for x in r["events"][0]["payments"]}
177        holder10 = [x["holder"] for x in r["events"][0]["positions"] if x["position"] == 10]
178        check("a holder with 10 bonds on the record date gets 500 per coupon",
179              bool(holder10) and Decimal(first[holder10[0]]["amount"]) == 500 * unit)
180
181    print(f"\n{'ALL CHECKS PASSED' if not failures else f'{len(failures)} FAILED: ' + '; '.join(failures)}"
182          f"  ({len(r['transactions'])} transactions)")
183    return 0 if not failures else 1
184
185
186if __name__ == "__main__":
187    sys.exit(main(sys.argv[1]))
