1// Operator (registrar + keeper) for bond_servicing: runs one bond's whole life, jumping the clock between record
2// moments on Surfpool or waiting for the real clock on devnet, and writes a report of every position, payment and
3// transaction.
4// Usage: npx tsx src/scenario.ts <scenarios/NAME.json> <report.json> <rpc url> <localnet|devnet>
5//   localnet (Surfpool): the clock is moved forward with surfnet_timeTravel.
6//   devnet: the real chain clock is awaited (bond time is compressed by the scenario's chainSecondsPerBondDay).
7// Logic: docs/bond-processing.md; overview: docs/technical-overview.md.

111  // Public RPC drops connections now and then ("fetch failed"; it stopped the second 1000-holder run): retry the same
112  // HTTP request until it gets through, waiting at most one bond day of chain time between tries (10 s for 429). Whether the run is
113  // still on schedule after an outage is decided by the phase guard, not here. Rate limits (429) are retried by
114  // web3.js itself. Resending a signed transaction is safe: same signature.
115  const retryingFetch = async (input: any, init?: any) => {
116    for (let attempt = 1; ; attempt++) {
117      let failure: string;
118      try {
119        const res = await fetch(input, init);
120        // Devnet answers 429 "Connection rate limits exceeded" in bursts (it stopped the third 1000-holder run after
121        // web3.js's own five retries): treat it like a dropped connection.
122        if (res.status !== 429) return res;
123        failure = "429";
124      } catch (e) {
125        failure = String((e as any)?.cause?.code ?? e);
126      }
127      console.log(`RPC request failed (${failure}), retry ${attempt}`);
128      // 429: back off up to devnet's 10-second rate-limit window. Retrying every chain second (1 bond day in the hostile
129      // run) with a few requests in flight kept this IP locked out for over 5 minutes (2026-10-10); a 10-second pause
130      // cleared it.
131      const capSeconds = failure === "429" ? Math.max(10, spec.chainSecondsPerBondDay) : spec.chainSecondsPerBondDay;
132      await new Promise((r) => setTimeout(r, 1_000 * Math.min(2 ** attempt, capSeconds)));
133    }

165      }

180    for (let attempt = 1; attempt <= 4; attempt++) {
181      try {

183      } catch (e: any) {

187          if (st && !st.err && (st.confirmationStatus === "confirmed" || st.confirmationStatus === "finalized")) return e.signature as string;
188        }
189        console.log(`${what}: not landed (attempt ${attempt}), sending again`);
190      }
191    }
192    throw new Error(`${what}: not landed after 4 attempts`);

409          label = `sell_to_offer ${o.id}: ${act.holder} sells ${quantity}`;
410          signature = await land("rpc", () => program.methods.sellToOffer(new BN(quantity))
411            .accountsPartial({ retirement: retirementAccounts(o, act.holder, owner), seller: owner, offer: offerPda(o.id) })
412            .signers([holders[act.holder]]).rpc());
413        } else {
414          const withdraw = !act.tender;
415          quantity = (act.tender ?? act.withdraw)!;
416          // The registrar tops up the holder's tender record deposit in the same transaction (one transaction per
417          // tender, not two).
418          const have = await connection.getBalance(owner);
419          const ixs: anchor.web3.TransactionInstruction[] = have < tenderDeposit
420            ? [SystemProgram.transfer({ fromPubkey: registrar.publicKey, toPubkey: owner, lamports: tenderDeposit - have })] : [];
421          ixs.push(await program.methods.changeTender(new BN(quantity), withdraw)
422            .accountsPartial({ owner, bond, offer: offerPda(o.id), holder: holderRecord(act.holder), tender: pda(Buffer.from("tender"), offerPda(o.id).toBuffer(), owner.toBuffer()) })
423            .instruction());
424          label = `change_tender ${o.id}: ${act.holder} ${withdraw ? "withdraws" : "tenders"} ${quantity}${ixs.length > 1 ? " (registrar funds the deposit)" : ""}`;
425          signature = await land("rpc", () => sendAndConfirmTransaction(connection, new Transaction().add(...ixs), [registrar, holders[act.holder]], { commitment: "confirmed" }));
426        }
427        await record(label, signature);
428        const sold = (await paidEvents(signature)).find((x) => x.name === "BondsBoughtBack");
429        offerReports[i].actions.push({
430          holder: act.holder, action: act.sell ? "sell" : act.tender ? "tender" : "withdraw", quantity, signature,
431          ...(sold ? { burned: Number(sold.data.burned), paid: sold.data.paid.toString() } : {}),
432        });
433      }
434    }
435  };
436  // Tender offers whose window closes before the next record moment are settled in this phase, by anyone.
437  const settleTenders = async (e: number) => {
438    const next = e + 1 < periods ? recordTimeOf(e + 1) : Infinity;
439    for (const [i, o] of spec.offers.entries()) {
440      if (o.kind !== "tender" || offerReports[i].settled || offerTimes(o).close >= next) continue;
441      await reachChainTime(chainTimeFor(offerTimes(o).close));
442      const tendered = new Map<string, number>();
443      for (const a of o.actions) tendered.set(a.holder, (tendered.get(a.holder) ?? 0) + (a.tender ?? 0) - (a.withdraw ?? 0));
444      // Tenders are resolved in the order they were made (the order of each holder's first tender, as sent above).
445      // Resolving needs no cash, so all are resolved first: a holder whose cash cannot move blocks nobody else.
446      const tenderPda = (name: string) => pda(Buffer.from("tender"), offerPda(o.id).toBuffer(), holders[name].publicKey.toBuffer());
447      for (const name of tendered.keys()) {
448        await record(`resolve_tender ${o.id} ${name}`, await land("rpc", () => program.methods.resolveTender()
449          .accountsPartial({ bond, offer: offerPda(o.id), holder: holderRecord(name), tender: tenderPda(name) }).rpc()));
450      }
451      for (const [name, q] of tendered) {
452        if (q <= 0) continue;
453        const signature = await land("rpc", () => program.methods.settleTender()
454          .accountsPartial({ retirement: retirementAccounts(o, name, registrar.publicKey), offer: offerPda(o.id), tender: pda(Buffer.from("tender"), offerPda(o.id).toBuffer(), holders[name].publicKey.toBuffer()) })
455          .rpc());
456        await record(`settle_tender ${o.id} ${name}`, signature);
457        const sold = (await paidEvents(signature)).find((x) => x.name === "BondsBoughtBack");
458        offerReports[i].settlements.push({
459          holder: name, tendered: q, burned: sold ? Number(sold.data.burned) : 0, paid: sold ? sold.data.paid.toString() : "0", signature,
460        });
461      }
462      offerReports[i].settled = true;
463    }
464  };
465  // One coupon: open, fund, confirm, then pay every holder with a position at the record moment.
466  const serviceCoupon = async (e: number, sv: Servicer) => {
467    const bondAccountNow: any = await program.account.bond.fetch(bond);
468    const recordTime = dayToUnix(Number(bondAccountNow.recordDays[e])) + Number(bondAccountNow.recordSecondsLocal) - Number(bondAccountNow.utcOffsetSeconds);
469    const paymentTime = dayToUnix(Number(bondAccountNow.paymentDays[e])) + Number(bondAccountNow.paymentSecondsLocal) - Number(bondAccountNow.utcOffsetSeconds);
470    const { event, vault, opened } = await openAndFund(e, `coupon ${e}`, sv);
471    vaults.push(vault.toBase58());
472    await reachChainTime(chainTimeFor(paymentTime));
473    const positions: any[] = [];
474    const payments: any[] = [];
475    const recs = await records(holderNames);
476    const bondBalances = await balances(holderNames.map(bondsAta));
477    holderNames.forEach((name, i) => positions.push({ holder: name, position: positionOf(recs[i], e), balanceNow: Number(bondBalances[i]) }));
478    const payable = holderNames.filter((_, i) => positionOf(recs[i], e) > 0);
479    const pay = async (names: string[]) => {
480      for (const batch of chunks(names, 6)) {
481        const ixs: anchor.web3.TransactionInstruction[] = [computeLimit];
482        for (const name of batch)
483          ixs.push(await sv.program.methods.payCoupon()
484            .accountsPartial({ payer: sv.payer, bond, event, holder: holderRecord(name), owner: holders[name].publicKey, vault, cashMint, holderCash: cashAta(name), memoProgram: null, cashTokenProgram: TOKEN_PROGRAM_ID })
485            .instruction());
486        const signature = await land("pay_coupon", () => sv.program.provider.sendAndConfirm!(new Transaction().add(...ixs), [], { commitment: "confirmed" }));
487        await record(`pay_coupon ${e} ${span(batch)}${sv.tag}`, signature);
488        for (const ev of (await paidEvents(signature)).filter((x) => x.name === "CouponPaid"))
489          payments.push({ holder: nameOf.get(ev.data.owner.toBase58()), amount: ev.data.amount.toString(), position: Number(ev.data.position), signature, payer: sv.payer.toBase58() });
490      }
491    };
492    // A holder the bank froze is paid after everyone else, once thawed.
493    const heldBack = (await h?.beforePayments(e)) ?? [];
494    await pay(payable.filter((n) => !heldBack.includes(n)));
495    await h?.afterPayments(e);
496    await pay(payable.filter((n) => heldBack.includes(n)));
497    events.push({
498      index: e, kind: "coupon", paymentDate: schedule.payment_dates[e], recordDate: schedule.record_dates[e],
499      recordTime: iso(recordTime), supplyAtRecord: Number(opened.supplyAtRecord), budget: opened.budget.toString(),
500      vault: vault.toBase58(), vaultLeft: (await balance(vault)).toString(), positions, payments, servicedBy: sv.payer.toBase58(),
501    });
502  };
503  const deferred: number[] = [];
504
505  // Coupons: at each record moment, the scenario's trades first (the market stays open), then the payments.
506  for (let e = 0; e < periods; e++) {
507    const bondAccountNow: any = await program.account.bond.fetch(bond);
508    const recordTime = dayToUnix(Number(bondAccountNow.recordDays[e])) + Number(bondAccountNow.recordSecondsLocal) - Number(bondAccountNow.utcOffsetSeconds);
509    await reachChainTime(chainTimeFor(recordTime) + (cluster === "devnet" ? 2 : 60));
510    // The scenario's actions for this phase must land before the next record moment, or positions would differ from
511    // the scenario's; stop instead of producing a run that does not match its own input.
512    const phaseEnd = e + 1 < periods ? chainTimeFor(dayToUnix(Number(bondAccountNow.recordDays[e + 1])) + Number(bondAccountNow.recordSecondsLocal) - Number(bondAccountNow.utcOffsetSeconds)) : Infinity;
513    const inPhase = async (what: string) => {
514      if ((await chainNow()) >= phaseEnd) throw new Error(`phase ${e} overran the next record moment before: ${what}`);
515    };
516    for (const t of spec.trades.filter((t) => t.afterRecordMoment === e)) {
517      await inPhase(`transfer ${t.from} -> ${t.to}`);
518      const ix = await createTransferCheckedWithTransferHookInstruction(
519        staticLookups, bondsAta(t.from), mint.publicKey, bondsAta(t.to), holders[t.from].publicKey, BigInt(t.amount), 0, [], "confirmed", TOKEN_2022_PROGRAM_ID,
520      );
521      await send(`transfer ${t.amount} ${t.from} -> ${t.to} (after record moment ${e})`, [ix], [registrar, holders[t.from]]);
522    }
523    await h?.afterRecordMoment(e);
524    await offerActions(e);
525    await h?.afterOfferActions(e);
526    await inPhase("offer actions");
527    if (!spec.handover || e <= spec.handover.operatorStopsAfterCoupon) await serviceCoupon(e, operator);
528    else deferred.push(e);
529    await settleTenders(e);
530  }
531
532  // After a handover the outside keeper services, late, every coupon the operator left.
533  for (const e of deferred) await serviceCoupon(e, outside!);
534
535  // Redemption: trading closed at the final record moment; burn and pay each holder.
536  const r = periods;
537  const redeemer = spec.handover ? outside! : operator;
538  await h?.afterFinalRecordMoment();
539  const { event: redemption, vault: redemptionVault, opened: redemptionOpened } = await openAndFund(r, "redemption", redeemer);
540  vaults.push(redemptionVault.toBase58());
541  await h?.beforeRedemption(redemption, redemptionVault);
542  const redemptions: any[] = [];
543  const finalRecs = await records(holderNames);
544  const redeemable = holderNames.filter((_, i) => Number(finalRecs[i].position.quantity) > Number(finalRecs[i].locked));
545  for (const batch of chunks(redeemable, 4)) {
546    const ixs: anchor.web3.TransactionInstruction[] = [computeLimit];
547    for (const name of batch)
548      ixs.push(await redeemer.program.methods.redeemHolder()
549        .accountsPartial({ payer: redeemer.payer, bond, mint: mint.publicKey, event: redemption, holder: holderRecord(name), owner: holders[name].publicKey, holderBonds: bondsAta(name), vault: redemptionVault, cashMint, holderCash: cashAta(name), memoProgram: null, tokenProgram: TOKEN_2022_PROGRAM_ID, cashTokenProgram: TOKEN_PROGRAM_ID })
550        .instruction());
551    const signature = await land("redeem", () => redeemer.program.provider.sendAndConfirm!(new Transaction().add(...ixs), [], { commitment: "confirmed" }));
552    await record(`redeem_holder ${span(batch)}${redeemer.tag}`, signature);
553    for (const ev of (await paidEvents(signature)).filter((x) => x.name === "BondsRedeemed"))
554      redemptions.push({ holder: nameOf.get(ev.data.owner.toBase58()), burned: Number(ev.data.burned), principal: ev.data.principal.toString(), signature });
555  }
556  await h?.afterRedemption(redemption, redemptionVault);
557  const supply = BigInt((await connection.getTokenSupply(mint.publicKey)).value.amount);
558
559  const report = {
560    scenario: key,
561    name: spec.name,
562    programId: program.programId.toBase58(),
563    bond: bond.toBase58(),
564    bondMint: mint.publicKey.toBase58(),
565    cashMint: cashMint.toBase58(),
566    terms: {
567      issue: spec.terms.issue, maturity: spec.terms.maturity, monthsPerPeriod: spec.terms.monthsPerPeriod, faceUnits: spec.terms.faceUnits,
568      cashDecimals: spec.terms.cashDecimals, rateBps: spec.terms.rateBps, basis: `FixedPerPeriod(${spec.terms.periodsPerYear})`, redemptionPriceBps: spec.terms.redemptionPriceBps,
569      paymentDates: schedule.payment_dates, recordDates: schedule.record_dates, recordTimeLocal: spec.terms.recordTimeLocal,
570      paymentOpensLocal: spec.terms.paymentOpensLocal, utcOffsetHours: spec.terms.utcOffsetHours,
571      calendar: spec.calendar, weekend: spec.weekend,
572    },
573    placements: spec.placements,
574    trades: spec.trades,
575    offerSpecs: spec.offers,
576    handover: spec.handover,
577    events,
578    offers: await Promise.all(offerReports.map(async (o) => ({ ...o, vaultLeft: (await balance(new PublicKey(o.vault))).toString() }))),
579    redemption: {
580      supplyAtRecord: Number(redemptionOpened.supplyAtRecord), budget: redemptionOpened.budget.toString(),
581      vault: redemptionVault.toBase58(), vaultLeft: (await balance(redemptionVault)).toString(), redemptions,
582    },
583    initialCash,
584    final: {
585      supply: supply.toString(),
586      vaults: Object.fromEntries(await Promise.all(vaults.map(async (v) => [v, (await balance(new PublicKey(v))).toString()]))),
587      cash: Object.fromEntries((await balances(holderNames.map(cashAta))).map((b, i) => [holderNames[i], b.toString()])),
588    },
589    transactions: log,
590    attempts: h?.attempts ?? [],
591  };
592  fs.mkdirSync(path.dirname(out), { recursive: true });
593  fs.writeFileSync(out, JSON.stringify(report, null, 1) + "\n");
594  console.log(`report: ${out} (${log.length} transactions)`);
595  if (h) {
596    const failed = h.attempts.filter((a) => !a.ok);
597    console.log(`attacks: ${h.attempts.length - failed.length} of ${h.attempts.length} refused by their guard`);
598    if (failed.length) throw new Error(`attacks not refused by their guard: ${failed.map((a) => `${a.id} (${a.outcome}${a.reason ? `: ${a.reason}` : ""})`).join("; ")}`);
599  }
600}
601
602main().catch((e) => {
603  console.error(e);
604  process.exit(1);
605});
