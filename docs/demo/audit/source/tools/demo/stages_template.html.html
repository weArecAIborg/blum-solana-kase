191// Text that wraps by words at maxW instead of running out of the picture; returns the y of its last line.
192function para(g, x, y, s, maxW, o = {}) {
193  const lh = (o.size ?? 12) * 1.35, t = label(g, x, y, "", o);
194  let words = [], row = t.append("tspan").attr("x", x), lines = 1;
195  for (const word of String(s).split(" ")) {
196    row.text([...words, word].join(" "));
197    if (words.length && row.node().getComputedTextLength() > maxW) {
198      row.text(words.join(" "));
199      row = t.append("tspan").attr("x", x).attr("dy", lh).text(word);
200      words = [word];
201      lines++;
202    } else words.push(word);
203  }
204  return y + (lines - 1) * lh;
205}
