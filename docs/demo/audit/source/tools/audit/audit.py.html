39def b58encode(raw: bytes) -> str:
40    n = int.from_bytes(raw, "big")
41    out = ""
42    while n:
43        n, r = divmod(n, 58)
44        out = B58[r] + out
45    return "1" * (len(raw) - len(raw.lstrip(b"\0"))) + out

48def b58decode(text: str) -> bytes:
49    n = 0
50    for ch in text:
51        n = n * 58 + B58.index(ch)
52    raw = n.to_bytes((n.bit_length() + 7) // 8, "big")
53    return b"\0" * (len(text) - len(text.lstrip("1"))) + raw

67class Rpc:
68    """Raw JSON-RPC client. Confirmed transactions do not change, so `getTransaction` replies are kept on disk
69    (ops/out/rpc-cache/<host>/), and measuring the same history again does not download it again.
70    A cached reply is used only if its slot is the slot the ledger reports now for that signature: fresh local
71    networks at the same URL replay identical transactions with identical signatures in other slots (measured
72    2026-10-08: 186 of 243 cached Surfpool replies carried the previous network's slot and block time)."""
73
74    def __init__(self, url):
75        self.url = url
76        host = urllib.parse.urlparse(url).netloc.replace(":", "_")
77        self.cache = ROOT / "ops/out/rpc-cache" / host
78
79    def call(self, method, params, slot=None):
80        if method == "getTransaction":
81            path = self.cache / f"{params[0]}.json"
82            if path.exists():
83                cached = json.loads(path.read_text())
84                if cached["slot"] == slot:
85                    return cached
86            result = self._call(method, params)
87            if result is not None:
88                self.cache.mkdir(parents=True, exist_ok=True)
89                path.write_text(json.dumps(result))
90            return result
91        return self._call(method, params)
92
93    def _call(self, method, params):
94        """JSON-RPC call. Public RPC nodes rate-limit (HTTP 429) and sometimes time out: wait and retry the same
95        request, up to 8 times."""
96        body = json.dumps({"jsonrpc": "2.0", "id": 1, "method": method, "params": params}).encode()
97        for attempt in range(8):
98            req = urllib.request.Request(self.url, body, {"Content-Type": "application/json"})
99            try:
100                with urllib.request.urlopen(req, timeout=60) as resp:
101                    reply = json.loads(resp.read())
102            except urllib.error.HTTPError as e:
103                if e.code != 429:
104                    raise
105                time.sleep(2 ** attempt)
106                continue
107            except (urllib.error.URLError, TimeoutError) as e:
108                if not isinstance(getattr(e, "reason", e), TimeoutError) and not isinstance(e, TimeoutError):
109                    raise
110                time.sleep(2 ** attempt)
111                continue
112            if "error" in reply and reply["error"].get("code") == 429:
113                time.sleep(2 ** attempt)
114                continue
115            if "error" in reply:
116                raise RuntimeError(f"{method}: {reply['error']}")
117            return reply["result"]
118        raise RuntimeError(f"{method}: still rate-limited or timing out after 8 attempts")

121def measure_state(rpc, mint, program, source, history_accounts=()):
122    """Channel S: every token account of the mint (owner, amount) as parsed by the node, and the supply.
123
124    source = "program-accounts": getProgramAccounts filtered by mint (complete; local nodes allow it).
125    source = "history-accounts": getMultipleAccounts over every token account seen in channel H (public devnet RPC
126    rate-limits getTokenLargestAccounts persistently). Completeness again rests on M1 and M3.
127    source = "largest-accounts": getTokenLargestAccounts + getMultipleAccounts (public devnet RPC refuses the
128    program-wide scan). It returns at most 20 accounts and may omit zero balances; M1 (sum = supply) and M3 (history
129    of every account seen) prove nothing with a balance was missed.
130    """
131    state = {}
132    if source == "program-accounts":
133        accounts = rpc.call("getProgramAccounts", [program, {"encoding": "jsonParsed",
134                                                            "filters": [{"memcmp": {"offset": 0, "bytes": mint}}]}])
135        parsed_list = [(a["pubkey"], a["account"]["data"]["parsed"]) for a in accounts]
136    elif source == "largest-accounts":
137        addresses = [a["address"] for a in rpc.call("getTokenLargestAccounts", [mint])["value"]]
138        infos = rpc.call("getMultipleAccounts", [addresses, {"encoding": "jsonParsed"}])["value"] if addresses else []
139        parsed_list = [(addr, info["data"]["parsed"]) for addr, info in zip(addresses, infos) if info]
140    elif source == "history-accounts":
141        # Every token account the runtime recorded in channel H, read now. Tokens held anywhere else would make
142        # M1 (sum = supply) fail; a change history missed would make M3 fail.
143        addresses = sorted(history_accounts)
144        parsed_list = []
145        for i in range(0, len(addresses), 100):
146            batch = addresses[i:i + 100]
147            infos = rpc.call("getMultipleAccounts", [batch, {"encoding": "jsonParsed"}])["value"]
148            parsed_list += [(addr, info["data"]["parsed"]) for addr, info in zip(batch, infos) if info]
149    else:
150        raise ValueError(f"unknown state source {source}")
151    for pubkey, parsed in parsed_list:
152        if parsed["type"] != "account" or parsed["info"]["mint"] != mint:
153            continue
154        state[pubkey] = {"owner": parsed["info"]["owner"], "amount": int(parsed["info"]["tokenAmount"]["amount"])}
155    supply = int(rpc.call("getTokenSupply", [mint])["value"]["amount"])
156    return state, supply

159def signatures(rpc, address):
160    sigs, before = [], None
161    while True:
162        opts = {"limit": 1000, "commitment": "confirmed"}
163        if before:
164            opts["before"] = before
165        page = rpc.call("getSignaturesForAddress", [address, opts])
166        sigs += page
167        if len(page) < 1000:
168            return sigs
169        before = page[-1]["signature"]

172def measure_history(rpc, addresses):
173    """Channel H: every transaction touching the addresses, with runtime-recorded token balances before/after.
174
175    The addresses are the two mints plus the bond's authority and its vaults (all public). The union matters: on
176    2026-10-08 public devnet RPC returned the mints' histories with a contiguous block of 60 finalized transactions
177    missing, while the vault's history returned them. A transaction no address lists would still be caught by M3."""
178    seen = {}
179    for address in addresses:
180        for s in signatures(rpc, address):
181            seen[s["signature"]] = s
182    txs = []
183    for sig in seen:
184        tx = rpc.call("getTransaction", [sig, {"encoding": "jsonParsed", "maxSupportedTransactionVersion": 0,
185                                               "commitment": "confirmed"}], seen[sig]["slot"])
186        keys = [k["pubkey"] for k in tx["transaction"]["message"]["accountKeys"]]
187        meta = tx["meta"]
188        balances = defaultdict(lambda: {"pre": 0, "post": 0, "owner": None, "mint": None})
189        for side in ("pre", "post"):
190            for b in meta[f"{side}TokenBalances"]:
191                entry = balances[keys[b["accountIndex"]]]
192                entry[side] = int(b["uiTokenAmount"]["amount"])
193                entry["owner"], entry["mint"] = b.get("owner"), b["mint"]
194        minted, burned = defaultdict(int), defaultdict(int)
195        instructions = list(tx["transaction"]["message"]["instructions"])
196        for inner in meta.get("innerInstructions") or []:
197            instructions += inner["instructions"]
198        for ix in instructions:
199            parsed = ix.get("parsed")
200            if ix.get("programId") == TOKEN_2022 and "data" in ix and not parsed:
201                raw = b58decode(ix["data"])
202                if len(raw) >= 10 and raw[0] == PERMISSIONED_BURN and raw[1] in (1, 2):
203                    burned[ix["accounts"][1]] += int.from_bytes(raw[2:10], "little")
204                continue
205            if not isinstance(parsed, dict) or ix.get("program") not in ("spl-token", "spl-token-2022"):
206                continue
207            info = parsed.get("info", {})
208            amount = info.get("amount") or (info.get("tokenAmount") or {}).get("amount")
209            kind = parsed["type"].lower()
210            if kind in ("mintto", "minttochecked"):
211                minted[info["mint"]] += int(amount)
212            elif "burn" in kind and "mint" in info and amount is not None:
213                burned[info["mint"]] += int(amount)
214        txs.append({
215            "signature": sig, "slot": tx["slot"], "blockTime": tx["blockTime"], "failed": meta["err"] is not None,
216            "deltas": {acct: {"owner": b["owner"], "mint": b["mint"], "delta": b["post"] - b["pre"]}
217                       for acct, b in balances.items()},
218            "minted": dict(minted), "burned": dict(burned),
219        })
220    txs.sort(key=lambda t: (t["slot"], t["blockTime"] or 0))
221    return txs

232def timeline(placements, trades, offers, record_times):
233    """Positions over the bond's life from the scenario alone, in the operator's order. After each record moment k:
234    the trades of phase k, then the offer actions of phase k (buyback sales retire bonds at once; tenders and
235    withdrawals only count). A tender offer is settled in the first phase whose next record moment comes after the
236    offer's close: everything tendered if the offer is not oversubscribed, else, in tender order, each tender's
237    running-total pro-rata share minus that of the tenders before it.
238    Returns the positions at each record moment, the final positions, each holder's bond movements and retirements
239    (burned, cash) in ledger order, and the offers' total budget."""
240    pos = defaultdict(int, placements)
241    moves = defaultdict(list)
242    for h, n in placements.items():
243        moves[h].append(n)
244    retire = defaultdict(list)
245    at_record = []
246    tendered = {o["id"]: {} for o in offers}
247    settled = set()
248
249    def close_time(o):
250        return record_times[o["close"]["recordMoment"]] + o["close"]["plusDays"] * 86_400
251
252    def retire_now(h, q, price):
253        pos[h] -= q
254        moves[h].append(-q)
255        retire[h].append((q, q * price))
256
257    for k in range(len(record_times)):
258        at_record.append(dict(pos))
259        for tr in [x for x in trades if x["afterRecordMoment"] == k]:
260            pos[tr["from"]] -= tr["amount"]
261            pos[tr["to"]] += tr["amount"]
262            moves[tr["from"]].append(-tr["amount"])
263            moves[tr["to"]].append(tr["amount"])
264        for o in offers:
265            for a in [x for x in o["actions"] if x["afterRecordMoment"] == k]:
266                if "sell" in a:
267                    retire_now(a["holder"], a["sell"], int(o["priceUnits"]))
268                else:
269                    t = tendered[o["id"]]
270                    t[a["holder"]] = t.get(a["holder"], 0) + a.get("tender", 0) - a.get("withdraw", 0)
271        nxt = record_times[k + 1] if k + 1 < len(record_times) else float("inf")
272        for o in offers:
273            if o["kind"] != "tender" or o["id"] in settled or close_time(o) >= nxt:
274                continue
275            t = tendered[o["id"]]
276            total, mx = sum(t.values()), o["maxQuantity"]
277            # The program resolves tenders in the order they were made (each holder's first tender) and gives each the
278            # running-total pro-rata share minus what the earlier ones received, so the offer buys exactly its maximum.
279            before = 0
280            for h, q in t.items():
281                accepted = q if total <= mx else (before + q) * mx // total - before * mx // total
282                before += q
283                if q > 0 and accepted > 0:
284                    retire_now(h, accepted, int(o["priceUnits"]))
285            settled.add(o["id"])
286    budgets = sum(int(o["priceUnits"]) * o["maxQuantity"] for o in offers)
287    return {"at_record": at_record, "final": dict(pos), "moves": moves, "retire": retire, "offer_budgets": budgets}

290def calculate(scn):
291    import QuantLib as ql
292    from financepy.products.bonds.bond import Bond as FpBond, CouponType
293    from financepy.utils.date import Date as FpDate
294    from financepy.utils.day_count import DayCountTypes
295    from financepy.utils.frequency import FrequencyTypes
296
297    t = scn["terms"]
298    face_units = int(t["faceUnits"])
299
300    def qd(s):
301        y, m, d = map(int, s.split("-"))
302        return ql.Date(d, m, y)
303
304    cal = ql.BespokeCalendar("bond")
305    for day in scn["weekend"]:
306        cal.addWeekend(getattr(ql, day))
307    for line in (ROOT / scn["calendar"]).read_text().splitlines():
308        if line and not line.startswith("#"):
309            cal.addHoliday(qd(line))
310    sched = ql.Schedule(qd(t["issue"]), qd(t["maturity"]), ql.Period(t["monthsPerPeriod"], ql.Months), cal,
311                        ql.Unadjusted, ql.Unadjusted, ql.DateGeneration.Backward, False)
312    pay_dates = [cal.adjust(sched[i], ql.Following) for i in range(1, len(sched))]
313    lag = t["recordBusinessDaysBeforePayment"]
314    record_dates = [cal.advance(d, -lag, ql.Days) if lag else d for d in pay_dates]
315    pay_dates = [f"{d.year():04d}-{d.month():02d}-{d.dayOfMonth():02d}" for d in pay_dates]
316    record_dates = [f"{d.year():04d}-{d.month():02d}-{d.dayOfMonth():02d}" for d in record_dates]
317
318    freq = {1: FrequencyTypes.ANNUAL, 2: FrequencyTypes.SEMI_ANNUAL}[t["periodsPerYear"]]
319    y, m, d = map(int, t["issue"].split("-"))
320    y2, m2, d2 = map(int, t["maturity"].split("-"))
321    fp = FpBond(FpDate(d, m, y), FpDate(d2, m2, y2), t["rateBps"] / 10_000, freq, DayCountTypes.THIRTY_360_BOND,
322                cpn_type=CouponType.FIXED)
323    per_bond = [Decimal(str(f)) * face_units for f in fp.flow_amounts[1:]]
324
325    record_times = [record_moment_unix(d, t["recordTimeLocal"], t["utcOffsetHours"]) for d in record_dates]
326    tl = timeline(scn["placements"], scn["trades"], scn["offers"], record_times)
327    coupons = {h: [] for h in scn["wallets"]}  # (coupon index, amount)
328    budgets_total = tl["offer_budgets"]  # each event's budget: the amount for the whole supply, rounded down once
329    for k, amount_per_bond in enumerate(per_bond):
330        held = tl["at_record"][k]
331        budgets_total += int((Decimal(sum(held.values())) * amount_per_bond).to_integral_value(rounding=ROUND_FLOOR))
332        for h in scn["wallets"]:
333            if held.get(h, 0) > 0:
334                coupons[h].append((k, int((Decimal(held[h]) * amount_per_bond).to_integral_value(rounding=ROUND_FLOOR))))
335    price = Decimal(t["redemptionPriceBps"]) / 10_000
336    final = tl["final"]
337    budgets_total += int((Decimal(sum(final.values())) * face_units * price).to_integral_value(rounding=ROUND_FLOOR))
338    bond_moves = {h: list(tl["moves"][h]) for h in scn["wallets"]}
339    retirements = {h: list(tl["retire"][h]) for h in scn["wallets"]}
340    principal = {}
341    for h, pos in final.items():
342        if pos > 0:
343            principal[h] = int((Decimal(pos) * face_units * price).to_integral_value(rounding=ROUND_FLOOR))
344            bond_moves[h].append(-pos)
345            retirements[h].append((pos, principal[h]))
346    return {"pay_dates": pay_dates, "record_dates": record_dates, "budgets_total": budgets_total,
347            "coupons": coupons, "bond_moves": bond_moves, "principal": principal, "retirements": retirements,
348            "face_units": face_units}

351def bond_vaults(rpc, bond_mint, cash_mint):

361def main(scenario_path, rpc_url, state_source):
362    scn = json.loads(Path(scenario_path).read_text())
363    rpc = Rpc(rpc_url)
364    unit = 10 ** scn["terms"]["cashDecimals"]
365    wallets = {pubkey_of(f): name for name, f in scn["wallets"].items()}
366    bond_mint, cash_mint = pubkey_of(scn["keys"]["bondMint"]), pubkey_of(scn["keys"]["cashMint"])
367    print(f"{scn['name']}\nbond mint {bond_mint}\ncash mint {cash_mint}\nRPC {rpc_url}  (state source: {state_source})")
368
369    print("\nMeasure: channel S (state of all token accounts) and channel H (transaction history)")
370    authority, vaults = bond_vaults(rpc, bond_mint, cash_mint)
371    txs = measure_history(rpc, [bond_mint, cash_mint, authority, *vaults])
372    # Public devnet's address indexes can miss transactions (2026-10-09: the cash mint's index lacked transactions of
373    # an earlier bond paid in the same cash token, while the holders' own accounts listed them). Where the two
374    # channels disagree on an account, read that account's own index and add what the others did not list; repeat
375    # until no account is left to consult. What still disagrees afterwards fails M1/M3 below.
376    consulted, added = set(), 0
377    while True:
378        seen = {mint: {a for t in txs for a, d in t["deltas"].items() if d["mint"] == mint} for mint in (bond_mint, cash_mint)}
379        bond_state, bond_supply = measure_state(rpc, bond_mint, TOKEN_2022, state_source, seen[bond_mint])
380        cash_state, cash_supply = measure_state(rpc, cash_mint, TOKEN, state_source, seen[cash_mint])
381        summed = defaultdict(int)
382        for t in txs:
383            for acct, d in t["deltas"].items():
384                summed[(acct, d["mint"])] += d["delta"]
385        disagree = {acct for state, mint in ((bond_state, bond_mint), (cash_state, cash_mint))
386                    for acct in set(state) | {a for a, m in summed if m == mint}
387                    if summed[(acct, mint)] != state.get(acct, {"amount": 0})["amount"]} - consulted
388        if not disagree:
389            break
390        known = {t["signature"] for t in txs}
391        extra = [t for t in measure_history(rpc, sorted(disagree)) if t["signature"] not in known]
392        consulted |= disagree
393        added += len(extra)
394        txs = sorted(txs + extra, key=lambda t: (t["slot"], t["blockTime"] or 0))
395    print(f"  {len(bond_state)} bond accounts, {len(cash_state)} cash accounts, {len(txs)} transactions"
396          + (f" ({added} listed only by the own indexes of {len(consulted)} accounts where the channels disagreed)"
397             if consulted else ""))
398    bond_txs = [t for t in txs if any(d["mint"] == bond_mint for d in t["deltas"].values()) or bond_mint in t["minted"]
399                or bond_mint in t["burned"]]
400    creation = signatures(rpc, bond_mint)[-1]  # oldest transaction of the bond mint = its creation
401    creation_time = creation["blockTime"]
402
403    print("\nModel-free invariants")
404    for label, state, supply in (("bond", bond_state, bond_supply), ("cash", cash_state, cash_supply)):
405        total = sum(a["amount"] for a in state.values())
406        check("M1", f"{label}: sum of all account balances = supply", total == supply, f"{total} vs {supply}")
407    bad = []
408    for t in txs:
409        for mint in (bond_mint, cash_mint):
410            net = sum(d["delta"] for d in t["deltas"].values() if d["mint"] == mint)
411            # A failed transaction changes no balance, whatever mint or burn instructions it carried (refused attacks).
412            expected = 0 if t["failed"] else t["minted"].get(mint, 0) - t["burned"].get(mint, 0)
413            if net != expected:
414                bad.append(f"{t['signature'][:12]} {mint[:6]} net {net} vs mint-burn {expected}")
415    check("M2", "every transaction: balance changes = minted - burned", not bad, "; ".join(bad[:3]))
416    history_balance = defaultdict(int)
417    for t in txs:
418        for acct, d in t["deltas"].items():
419            history_balance[(acct, d["mint"])] += d["delta"]
420    # Every account seen in either channel: its history sum must equal its measured balance (0 if absent from S).
421    mismatched = [acct for state, mint in ((bond_state, bond_mint), (cash_state, cash_mint))
422                  for acct in set(state) | {a for a, m in history_balance if m == mint}
423                  if history_balance[(acct, mint)] != state.get(acct, {"amount": 0})["amount"]]
424    check("M3", "history sums = current balances (both channels agree)", not mismatched, ", ".join(mismatched[:3]))
425    # Many bonds may be paid in the same cash token and investors may hold several bonds, so every servicing check
426    # below is scoped to this bond's vaults (found above); M1-M3 above stay global over both mints.
427    vault_set = set(vaults)
428    check("M4", f"the bond's vaults = cash accounts owned by the bond token's mint authority ({authority[:8]}...)",
429          bool(vaults), f"{len(vaults)} vaults")
430    vault_delta = lambda t: sum(t["deltas"][v]["delta"] for v in vault_set if v in t["deltas"])
431    vault_txs = [t for t in txs if any(v in t["deltas"] and t["deltas"][v]["delta"] for v in vault_set)]
432    bad_in = [t["signature"][:12] for t in vault_txs
433              if any(t["deltas"][v]["delta"] > 0 for v in vault_set if v in t["deltas"]) and not t["minted"].get(cash_mint)]
434    check("M4", "cash enters the vaults only from the bank's mints", bool(vaults) and not bad_in, ", ".join(bad_in[:3]))
435    leaks = [f"{a} (owner {d['owner']})" for t in vault_txs if vault_delta(t) < 0
436             for a, d in t["deltas"].items() if d["mint"] == cash_mint and d["delta"] > 0 and a not in vault_set
437             and d["owner"] not in wallets]
438    check("M4", "cash leaves the vaults only to scenario wallets", bool(vaults) and not leaks, "; ".join(leaks[:3]))
439    bond_strangers = [f"{a} (owner {d['owner']})" for t in txs for a, d in t["deltas"].items()
440                      if d["mint"] == bond_mint and d["delta"] != 0 and d["owner"] not in wallets]
441    check("M4", "no account outside the scenario wallets moved bonds", not bond_strangers, "; ".join(bond_strangers[:3]))
442
443    print("\nMeasured vs calculated (exact integers, base units)")
444    calc = calculate(scn)
445    # Measured per holder, in ledger order.
446    cash_in = defaultdict(list)    # (tx, amount) coupon arrivals from this bond's vault
447    bond_moves = defaultdict(list)
448    retirements = defaultdict(list)  # holder -> [(burned, cash received in the same transaction)], ledger order
449    outside = defaultdict(list)    # holder cash movements in transactions that do not involve this bond's vault
450    vault_in = vault_out = 0
451    for t in txs:
452        touches_vault = any(v in t["deltas"] for v in vault_set)
453        from_vault = vault_delta(t) < 0
454        for acct, d in t["deltas"].items():
455            if d["delta"] == 0:
456                continue
457            holder = wallets.get(d["owner"])
458            if d["mint"] == bond_mint and holder:
459                bond_moves[holder].append(d["delta"])
460            if d["mint"] == cash_mint and acct in vault_set:
461                vault_in += max(d["delta"], 0)
462                vault_out += max(-d["delta"], 0)
463            if d["mint"] == cash_mint and holder and not touches_vault:
464                outside[holder].append(d["delta"])
465        if not from_vault:
466            continue
467        burned_by = {wallets[d["owner"]]: -d["delta"] for d in t["deltas"].values()
468                     if d["mint"] == bond_mint and d["delta"] < 0 and t["burned"].get(bond_mint) and d["owner"] in wallets}
469        for d in t["deltas"].values():
470            holder = wallets.get(d["owner"])
471            if d["mint"] == cash_mint and holder and d["delta"] > 0:
472                if holder in burned_by:
473                    retirements[holder].append((burned_by[holder], d["delta"]))
474                else:
475                    cash_in[holder].append((t, d["delta"]))
476
477    for h in scn["wallets"]:
478        expected = [amount for _, amount in calc["coupons"][h]]
479        measured = [amount for _, amount in cash_in[h]]
480        check("C1", f"{h}: coupon arrivals from the vault", measured == expected,
481              f"measured {[x / unit for x in measured]} vs calculated {[x / unit for x in expected]}")
482        check("C2", f"{h}: bond movements", bond_moves[h] == calc["bond_moves"][h],
483              f"measured {bond_moves[h]} vs calculated {calc['bond_moves'][h]}")
484        if calc["retirements"][h] or retirements[h]:
485            got = [(b, c / unit) for b, c in retirements[h]]
486            exp = [(b, c / unit) for b, c in calc["retirements"][h]]
487            check("C3", f"{h}: bonds retired against cash in one transaction (buyback, tender, redemption) = calculated",
488                  got == exp, f"measured (burned, cash) {got} vs calculated {exp}")
489    total_out = (sum(a for h in calc["coupons"] for _, a in calc["coupons"][h])
490                 + sum(c for h in calc["retirements"] for _, c in calc["retirements"][h]))
491    final_vaults = sum(cash_state.get(v, {"amount": 0})["amount"] for v in vault_set)
492    # Each event is funded with its budget: the amount for the whole supply, rounded down once; each offer with price x
493    # maximum. Holders are rounded down one by one and offers may not fill, so what stays in the vaults is the rounding
494    # difference plus unfilled offer cash, and nothing else.
495    leftover = calc["budgets_total"] - total_out
496    check("C4", "vaults: inflow = outflow + final balance; outflow = calculated payments; final = calculated leftover",
497          bool(vaults) and vault_in == vault_out + final_vaults and vault_out == total_out and final_vaults == leftover,
498          f"in {vault_in / unit}, out {vault_out / unit}, final {final_vaults / unit}, calculated out {total_out / unit}, "
499          f"calculated leftover {leftover / unit}")
500
501    # C7: per holder, what this bond measurably delivered (cash from its vault, bonds held now) against the terms.
502    # Holder cash moved outside this bond (other bonds paid in the same cash token, the holder's own transfers) is
503    # the holder's own business; it is shown, not judged.
504    owner_bonds = defaultdict(int)
505    for a in bond_state.values():
506        owner_bonds[a["owner"]] += a["amount"]
507    for pk, h in wallets.items():
508        expected_cash = sum(a for _, a in calc["coupons"][h]) + sum(c for _, c in calc["retirements"][h])
509        received = sum(a for _, a in cash_in[h]) + sum(c for _, c in retirements[h])
510        expected_bonds = sum(calc["bond_moves"][h])
511        check("C7", f"{h}: cash from this bond and bonds held now = calculated",
512              received == expected_cash and owner_bonds[pk] == expected_bonds,
513              f"cash {received / unit} vs {expected_cash / unit}, bonds {owner_bonds[pk]} vs {expected_bonds}"
514              + (f"; outside this bond: {len(outside[h])} cash movements, net {sum(outside[h]) / unit:+,.6f}"
515                 if outside[h] else ""))
516
517    # C5 timing. Bond time of a transaction = bond time at creation + elapsed chain time scaled by the scenario's
518    # compression (chain seconds per bond day), measured from the bond mint's creation transaction.
519    almaty = timezone(timedelta(hours=scn["terms"]["utcOffsetHours"]))
520    bond_origin = datetime.fromisoformat(scn["bondTimeAtCreation"]).timestamp()
521    scale = scn["chainSecondsPerBondDay"]
522    rec_h, rec_m = map(int, scn["terms"]["recordTimeLocal"].split(":"))
523    dl_h, dl_m = map(int, scn["terms"]["paymentDeadlineLocal"].split(":"))
524
525    def record_moment(k):
526        y, m, d = map(int, calc["record_dates"][k].split("-"))
527        return datetime(y, m, d, rec_h, rec_m, tzinfo=almaty).timestamp()
528
529    late = []
530    handover = scn["handover"]
531    for h in scn["wallets"]:
532        for (k, _), (t, _) in zip(calc["coupons"][h], cash_in[h]):
533            bond_ts = bond_origin + (t["blockTime"] - creation_time) * 86_400 / scale
534            bond_time = datetime.fromtimestamp(bond_ts, almaty)
535            if scale == 86_400:
536                local = bond_time.hour * 60 + bond_time.minute
537                ok = bond_time.date().isoformat() == calc["pay_dates"][k] and rec_h * 60 + rec_m <= local < dl_h * 60 + dl_m
538            elif handover is not None and k > handover["operatorStopsAfterCoupon"]:
539                # Declared operator failure: these coupons are paid late, after the final record moment.
540                ok = bond_ts >= record_moment(len(calc["pay_dates"]) - 1)
541            else:
542                nxt = record_moment(k + 1) if k + 1 < len(calc["pay_dates"]) else float("inf")
543                ok = record_moment(k) <= bond_ts < nxt
544            if not ok:
545                late.append(f"{h} coupon {k} at bond time {bond_time.isoformat()} (due {calc['pay_dates'][k]})")
546    if scale == 86_400:
547        check("C5", "every coupon paid on its payment date between 09:00 and 17:00 Almaty", not late, "; ".join(late[:3]))
548    else:
549        rule = ("before the next one" if handover is None else
550                f"before the next one; coupons after {handover['operatorStopsAfterCoupon']} after the final record moment "
551                f"(declared operator failure)")
552        check("C5", f"every coupon paid after its record moment and {rule} (compressed time: 1 bond day = "
553                    f"{scale} s, so the 8-hour 09:00-17:00 window is {scale / 3:.1f} s and is not checked)",
554              not late, "; ".join(late[:3]))
555
556    if 10 in scn["placements"].values():
557        h = next(name for name, n in scn["placements"].items() if n == 10)
558        first = cash_in[h][0][1] if cash_in[h] else None
559        check("C6", f"listing example: {h} held 10 bonds at the first record moment, measured first coupon = 500",
560              first == 500 * unit, f"{first / unit if first is not None else None}")
561
562    failed = [r for r in results if not r[2]]
563    last_audit["transactions"] = len(txs)
564    print(f"\n{'AUDIT PASSED' if not failed else f'AUDIT FAILED: {len(failed)} checks'}  "
565          f"({len(results)} checks, {len(txs)} transactions measured)")
566    return 0 if not failed else 1
